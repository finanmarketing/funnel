left join affiliate_configurations affc on affc.id = aff.affiliate_configuration_id
where aff.loan_application_id = lapp.id order by aff.entity_created desc limit 1)
as affc_id on true
left join lateral (select new_loan_price 
from affiliate_costs acost
where acost.affiliate_configuration_id = affc_id.id
and affc_id.partner_name = acost.partner_name 
and lag.sign_date >= acost.date_from and lag.sign_date <= acost.date_to
and lapp.loans_count = 0
order by acost.id desc limit 1)
as cost1 on true
left join lateral (select new_loan_price 
from affiliate_costs acost
where acost.affiliate_configuration_id = affc_id.id
and (acost.partner_name = '' or acost.partner_name is null)
and lag.sign_date >= acost.date_from and lag.sign_date <= acost.date_to
and lapp.loans_count = 0
order by acost.id desc limit 1)
as cost2 on true
where lapp.is_additional_amount_application = 'f'
and lapp.entity_created >= date_trunc('MONTH', now()-interval '3 month')
;
